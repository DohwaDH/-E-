    %% Sub-Systems grouped by file functions
    class SaveSystem {
        <<Utility>>
        +get_save_path() str
        +get_progress_save_path() str
        +write_progress_save(data: Dict) None
        +read_progress_save() Dict
        +clear_progress_save() None
    }

    class OptionSystem {
        <<Utility>>
        -String OPTIONS_FILE
        +load_options() Dict
        +save_options(data: Dict) None
    }

    class BitsManager {
        <<Utility>>
        -String BITS_FILE
        +load_bits_data() Dict
        +save_bits_data(data: Dict) None
        +get_bits_amount() int
        +add_bits(amount: int) int
        +spend_bits(amount: int) tuple
    }

    class AdSystem {
        <<Utility>>
        -String ADMOB_REWARDED_TEST_ID
        -String ADMOB_REWARDED_REAL_ID
        +init_mobile_ads() None
        +show_ether_reward_ad() tuple
        +show_revive_reward_ad() tuple
        +poll_android_ad_reward() String
        +poll_android_revive_reward() bool
        +grant_ether_ad_reward() tuple
    }

    class GoogleAuthSystem {
        <<Utility>>
        -String GOOGLE_AUTH_FILE
        +load_google_auth_data() Dict
        +save_google_auth_data(data: Dict) None
        +show_google_login() tuple
        +refresh_google_auth_state_from_bridge() String
        +get_google_auth_display_text() String
    }

    class SkinManager {
        <<Utility>>
        -String SKIN_DIR_NAME
        -int MAX_SKIN_SLOTS
        +load_skin_slots() Dict
        +save_skin_slots(data: Dict) None
        +apply_skin_slot(slot_index: int) tuple
        +unlock_next_skin_slot() tuple
        +download_skin_from_url(url: String, slot_index: int) tuple
        +get_current_skin_path() String
    }

    %% Kivy Game UI and App Structure
    class OrdealDungeonApp {
        +build() ScreenManager
        +on_start() None
    }

    class GameScreen {
        +Player player
        +Enemy current_enemy
        +int current_floor
        +explore() None
        +start_battle() None
        +use_skill(skill_key: String) None
        +use_potion() None
    }

    class SkinShopScreen {
        +update_slots_ui() None
        +buy_slot() None
        +download_new_skin() None
    }

    %% Relationships
    App <|-- OrdealDungeonApp : Inherits
    Screen <|-- GameScreen : Inherits
    Screen <|-- SkinShopScreen : Inherits

    OrdealDungeonApp ..> SaveSystem : Uses
    OrdealDungeonApp ..> OptionSystem : Uses
    OrdealDungeonApp ..> AdSystem : Initializes

    GameScreen ..> BitsManager : Updates Ether
    GameScreen ..> SaveSystem : Saves State
    GameScreen ..> AdSystem : Calls Revive Ad
    GameScreen ..> FloorData : Reads Enemy Data

    SkinShopScreen ..> SkinManager : Controls Slots
    SkinShopScreen ..> BitsManager : Spends Ether

    AdSystem ..> BitsManager : Rewards Ether
